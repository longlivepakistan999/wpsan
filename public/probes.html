添加探针
在探针服务器上运行以下命令来注册探针
建议格式: probe-{地区}-{序号}
建议根据服务器配置设置，默认为 10
# 1. 下载探针程序
git clone https://github.com/your-repo/wpsan.git
cd wpsan/probe
# 2. 配置连接信息
cp .env.example .env
vim .env
# 3. 启动探针
php probe.php
PROBE_ID=probe-tokyo-01
PROBE_CONCURRENCY=10
MASTER_REDIS_HOST=主服务器IP
MASTER_REDIS_PORT=6379
MASTER_MYSQL_HOST=主服务器IP
MASTER_MYSQL_DATABASE=wpsan
MASTER_MYSQL_USER=wpsan
MASTER_MYSQL_PASSWORD=密码
